0023: export async function createProspectCore(
0024:   payload: ProspectRegistrationPayload,
0025: ) {
0026:   const prospect = await supabaseDAL.createProspect({
0027:     full_name:            payload.fullName,
0028:     email:                payload.email,
0029:     country:              payload.country,
0030:     program_of_interest:  payload.programOfInterest,
0031:     phone:                payload.phone ?? null,
0032:   });
0033: 
0034:   const funnel = await supabaseDAL.getFunnelByCode(resolveRegistrationFunnel());
0035: 
0036:   if (funnel) {
0037:     await supabaseDAL.createProspectJourney({
0038:       prospect_id: prospect.id,
0039:       funnel_id:   funnel.id,
0040:       current_stage: 'registered',
0041:     });
0042:   }
0043: 
0044:   return prospect;
0045: }
0046:

0049: const DEFAULT_FUNNEL_CODE = "CTBC-2026";
0050:

0052:   return DEFAULT_FUNNEL_CODE;
0053: }

0055: export async function submitProspectRegistration(
0056:   payload: ProspectRegistrationPayload,
0057: ): Promise<ProspectRegistrationResult> {
0058:   const prospect = await createProspectCore(payload);
0059: 
0060: const email = registrationConfirmationTemplate({
0061:     fullName: payload.fullName,
0062:     program: payload.programOfInterest,
0063:   });
0064: 
0065:   await sendNotification({
0066:     recipientEmail: payload.email,
0067:     recipientId: prospect.id,
0068:     recipientType: "prospect",
0069:     type: "registration_confirmation",
0070:     subject: email.subject,
0071:     message: email.text,
0072:   });
0073: 
0074: 
0075:   await recordProspectActivity(
0076:     prospect.id,
0077:     "registration_confirmation_email_sent",
0078:     {
0079:       recipient: payload.email,
0080:       subject: "Thank you for registering your interest in CTBC University",
0081:       provider: "resend",
0082:       channel: "email",
0083:     },
0084:   ).catch((err) =>
0085:     console.error(
0086:       "[ProspectActivity] Failed to record email activity:",
0087:       err,
0088:     ),
0089:   );
0090: 
0091: 
0092: 
0093:   return {
0094:     accepted: true,
0095:     status: "validated",
0096:     message: "Prospect registration accepted for future persistence.",
0097:   };
0098: }
